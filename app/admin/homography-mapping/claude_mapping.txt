  /**
   * Universal coordinate transformer for both camera and map views
   * Handles object-fit: contain properly for consistent behavior
   */
  const getImageCoordinates = useCallback((event: React.MouseEvent, imageElement: HTMLImageElement, naturalWidth: number, naturalHeight: number) => {
    if (!imageElement || !naturalWidth || !naturalHeight) {
      console.error('Missing required parameters for coordinate transformation')
      return null
    }

    const rect = imageElement.getBoundingClientRect()
    
    // Calculate the actual scale used by object-fit: contain
    const scaleX = rect.width / naturalWidth
    const scaleY = rect.height / naturalHeight
    const actualScale = Math.min(scaleX, scaleY) // object-fit: contain uses minimum
    
    // Calculate the actual displayed image dimensions
    const displayedWidth = naturalWidth * actualScale
    const displayedHeight = naturalHeight * actualScale
    
    // Calculate centering offsets (letterboxing/pillarboxing)
    const offsetX = (rect.width - displayedWidth) / 2
    const offsetY = (rect.height - displayedHeight) / 2
    
    // Get click position relative to the container
    const clickX = event.clientX - rect.left
    const clickY = event.clientY - rect.top
    
    // Check if click is within the actual image area
    if (clickX < offsetX || clickX > offsetX + displayedWidth ||
        clickY < offsetY || clickY > offsetY + displayedHeight) {
      console.warn('Click outside image bounds')
      return null
    }
    
    // Convert to image coordinates
    const imageX = (clickX - offsetX) / actualScale
    const imageY = (clickY - offsetY) / actualScale
    
    // Clamp to image boundaries (safety check)
    const clampedX = Math.max(0, Math.min(naturalWidth, Math.round(imageX)))
    const clampedY = Math.max(0, Math.min(naturalHeight, Math.round(imageY)))
    
    return {
      x: clampedX,
      y: clampedY,
      scale: actualScale,
      offset: { x: offsetX, y: offsetY },
      withinBounds: true
    }
  }, [])

  /**
   * Convert image coordinates back to display coordinates for rendering overlays

  const getDisplayCoordinates = useCallback((imagePoint: Point, imageElement: HTMLImageElement | null, naturalWidth: number, naturalHeight: number) => {
    if (!imageElement || !naturalWidth || !naturalHeight) return imagePoint
    
    const rect = imageElement.getBoundingClientRect()
    const actualScale = Math.min(rect.width / naturalWidth, rect.height / naturalHeight)
    const displayedWidth = naturalWidth * actualScale
    const displayedHeight = naturalHeight * actualScale
    const offsetX = (rect.width - displayedWidth) / 2
    const offsetY = (rect.height - displayedHeight) / 2
    
    return {
      x: imagePoint.x * actualScale + offsetX,
      y: imagePoint.y * actualScale + offsetY
    }
  }, [])

  // ========================================
  // HOMOGRAPHY VALIDATION SYSTEM
  // ========================================

  /**
   * Comprehensive homography validation with quality scoring
   */
  const validateHomographyMapping = useCallback((srcPoints: Point[], dstPoints: Point[]): ValidationResult => {
    const validation: ValidationResult = {
      isValid: false,
      errors: [],
      warnings: [],
      quality: 0
    }

    // Check minimum points
    if (srcPoints.length < 4 || dstPoints.length < 4) {
      validation.errors.push('Minimum 4 points required for homography')
      return validation
    }

    // Check point correspondence
    if (srcPoints.length !== dstPoints.length) {
      validation.errors.push('Source and destination points count mismatch')
      return validation
    }

    // Check for collinear points
    const arePointsCollinear = (points: Point[], tolerance = 10): boolean => {
      if (points.length < 3) return false
      for (let i = 2; i < points.length; i++) {
        const area = Math.abs(
          (points[1].x - points[0].x) * (points[i].y - points[0].y) -
          (points[i].x - points[0].x) * (points[1].y - points[0].y)
        )
        if (area > tolerance) return false
      }
      return true
    }

    if (arePointsCollinear(srcPoints)) {
      validation.errors.push('Source points are collinear - invalid for homography')
      return validation
    }

    if (arePointsCollinear(dstPoints)) {
      validation.errors.push('Destination points are collinear - invalid for homography')
      return validation
    }

    // Check for duplicate points
    const hasDuplicatePoints = (points: Point[], tolerance = 5): boolean => {
      for (let i = 0; i < points.length; i++) {
        for (let j = i + 1; j < points.length; j++) {
          const distance = Math.sqrt(
            Math.pow(points[i].x - points[j].x, 2) +
            Math.pow(points[i].y - points[j].y, 2)
          )
          if (distance < tolerance) return true
        }
      }
      return false
    }

    if (hasDuplicatePoints(srcPoints) || hasDuplicatePoints(dstPoints)) {
      validation.warnings.push('Duplicate points detected - may affect accuracy')
    }

    // Calculate quality score based on point distribution
    const calculatePolygonArea = (points: Point[]): number => {
      if (points.length < 3) return 0
      let area = 0
      for (let i = 0; i < points.length; i++) {
        const j = (i + 1) % points.length
        area += points[i].x * points[j].y
        area -= points[j].x * points[i].y
      }
      return Math.abs(area) / 2
    }

    const srcArea = calculatePolygonArea(srcPoints)
    const dstArea = calculatePolygonArea(dstPoints)
    
    let score = 50 // Base score
    
    // Bonus for more points
    if (srcPoints.length > 4) {
      score += Math.min(20, (srcPoints.length - 4) * 5)
    }
    
    // Bonus for good coverage
    if (srcArea > 10000 && dstArea > 10000) {
      score += 20
    }
    
    // Calculate aspect ratio similarity
    const getBoundingBox = (points: Point[]) => {
      const xs = points.map(p => p.x)
      const ys = points.map(p => p.y)
      return {
        width: Math.max(...xs) - Math.min(...xs),
        height: Math.max(...ys) - Math.min(...ys)
      }
    }

    const srcBounds = getBoundingBox(srcPoints)
    const dstBounds = getBoundingBox(dstPoints)
    
    const srcAspect = srcBounds.width / srcBounds.height
    const dstAspect = dstBounds.width / dstBounds.height
    const aspectRatio = Math.min(srcAspect, dstAspect) / Math.max(srcAspect, dstAspect)
    
    score += aspectRatio * 10 // Aspect ratio similarity bonus

    // Small area penalty
    if (srcArea < 1000 || dstArea < 1000) {
      validation.warnings.push('Small mapping area may affect accuracy')
      score -= 10
    }

    validation.isValid = true
    validation.quality = Math.min(100, Math.max(0, Math.round(score)))

    if (validation.quality < 60) {
      validation.warnings.push('Low mapping quality - consider adding more points or improving distribution')
    }

    return validation
  }, [])

   */
  const detectCameraResolution = useCallback(async (camera: Camera) => {
    try {
      // Try to get resolution from camera metadata first
      if (camera.resolution) {
        setCameraResolution(camera.resolution)
        return camera.resolution

          width: img.naturalWidth,
          height: img.naturalHeight
        }
        setCameraResolution(resolution)
        return resolution
      }

      const fallback = { width: 1920, height: 1080 }
      setCameraResolution(fallback)
      return fallback
    } catch (error) {
      console.error('Could not detect camera resolution:', error)
      const fallback = { width: 1920, height: 1080 }
      setCameraResolution(fallback)
      return fallback
    }

    if (!isSelecting && !isMappingMode && currentStep !== 2) return
    if (isMappingMode && type !== selectionSequence && currentStep !== 2) return

    let naturalWidth: number, naturalHeight: number

      naturalHeight = mallMapNaturalHeight || img.naturalHeight

      x: coordinates.x,
      y: coordinates.y,
      id: Date.now(),
      timestamp: Date.now()

          }

        }
      }
    }

  // ========================================
  // UTILITY FUNCTIONS
  // ========================================

  const isPointInsideZone = useCallback((point: Point, zone: { points: Point[] }): boolean => {
    if (zone.points.length < 3) return false

    let inside = false
    for (let i = 0, j = zone.points.length - 1; i < zone.points.length; j = i++) {
      const xi = zone.points[i].x
      const yi = zone.points[i].y
      const xj = zone.points[j].x
      const yj = zone.points[j].y

      const intersect = ((yi > point.y) !== (yj > point.y)) &&
        (point.x < (xj - xi) * (point.y - yi) / (yj - yi) + xi)
      if (intersect) inside = !inside
    }
    return inside
  }, [])

    setTimeout(() => setTempError(null), 3000)
  }, [])

          console.warn("Could not fetch mall map:", error)
          setMallMapImage(null)
        }

          const camerasResponse = await fetch(`http://localhost:8000/mall/${user.mall_id}/cameras`, {
            headers: { "Authorization": `Bearer ${token}` }

            setCameras(Array.isArray(camerasData) ? camerasData : [])

        setMallMapImage(null)
      } finally {
        setIsLoading(false)
      }

        const data = await response.json()
        const zones = data.homography_map?.zones || []
        setZonesData(zones)
      } catch (error) {
        console.error("Error fetching camera data:", error)
        toast({
          title: "Error",
          description: error instanceof Error ? error.message : "Failed to fetch camera data",
          variant: "destructive"
        })

    fetchZonesData()

      setCalibrationQuality(validation.quality)
    } else {

      setCalibrationQuality(0)
    }
  }, [selectedObject, validateHomographyMapping])

  // Authentication check

    const token = localStorage.getItem("token")
    if (!token) {
      toast({
        title: "Authentication Required",
        description: "Please login to continue",
        variant: "destructive",
      })
      router.push("/login")
    }

  // Clean up object URLs

        URL.revokeObjectURL(cameraFrame)

  // ========================================
  // CAMERA OPERATIONS
  // ========================================

  const getCameraFrame = async (camera: Camera) => {
    try {
      setIsCapturing(true)
      setCaptureError(null)

      const token = localStorage.getItem("token")
      if (!token) {
        toast({
          title: "Authentication Required",
          description: "Please login to continue",
          variant: "destructive",
        })
        router.push("/login")
        return
      }

      const rtspUrl = `rtsp://${camera.username}:${camera.password}@${camera.ip_address}:554/cam/realmonitor?channel=1&subtype=0`

      const response = await fetch("http://localhost:8000/camera/frame", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          "Authorization": `Bearer ${token.trim()}`
        },
        body: JSON.stringify({ rtsp_url: rtspUrl })
      })

      if (response.status === 401) {
        toast({
          title: "Session Expired",
          description: "Your session has expired. Please login again.",
          variant: "destructive",
        })
        localStorage.removeItem("token")
        localStorage.removeItem("user")
        router.push("/login")
        return
      }

      if (!response.ok) {
        const contentType = response.headers.get("content-type")
        let errorMessage = "Failed to get camera frame"

        if (contentType && contentType.includes("application/json")) {
          try {
            const errorData = await response.json()
            if (errorData.detail && errorData.detail.includes("Unauthorized")) {
              errorMessage = "Camera authentication failed. Please check camera credentials."
            } else {
              errorMessage = errorData.detail || errorData.message || JSON.stringify(errorData)
            }
          } catch (e) {
            console.error("Error parsing JSON response:", e)
          }
        }

        setCaptureError(errorMessage)
        toast({
          title: "Error",
          description: errorMessage,
          variant: "destructive",
        })
        return
      }

      const blob = await response.blob()
      const objectUrl = URL.createObjectURL(blob)
      setCameraFrame(objectUrl)

      // Detect camera resolution after frame is loaded
      setTimeout(() => detectCameraResolution(camera), 100)
    } catch (error) {
      console.error("Error capturing frame:", error)
      setCaptureError(error instanceof Error ? error.message : "Failed to capture frame")
      toast({
        title: "Error",
        description: error instanceof Error ? error.message : "Failed to capture frame",
        variant: "destructive",
      })
    } finally {
      setIsCapturing(false)
    }
  }

  // ========================================

  }

  const handleZoneSubmit = (e: React.FormEvent) => {

  const handleCompleteZoneMapping = () => {

      setIsZoneMappingComplete(true)
      setCurrentStep(3)
      setCameraZonePoints([])
      setMapZonePoints([])
      setZoneMappingMode("camera")

        title: "Success",
        description: "Zone mapping completed",
      })
    } else {
      toast({
        title: "Error",
        description: "Please select at least 3 points on both views",
        variant: "destructive",
      })
    }
